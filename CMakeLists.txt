cmake_minimum_required(VERSION 3.21)

# webbed-synapses; `snps` for short, which is what the code and options use.
project(webbed-synapses
  VERSION 0.1.0
  DESCRIPTION "A WebRTC transport for Blizzard-era game networking"
  LANGUAGES C CXX)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")
include(SnpModule)

set(CMAKE_C_STANDARD 11)

option(SNP_BUILD_PROVIDERS "Build the SNP modules (32-bit Windows only)" ON)
option(SNP_BUILD_TESTS     "Build the ABI layout checks and unit tests"  ON)

# The real transport pulls in libdatachannel, Mbed TLS and usrsctp at
# configure time. A build that only checks struct layouts has no use for them.
option(SNPS_BUILD_TRANSPORT "Build the WebRTC transport core (fetches libdatachannel)" ON)

# caps.mpq is generated by tools that run *during* the build, so they have to
# be runnable by the build machine. A 32-bit tool on 64-bit Windows is fine --
# that is not cross-compiling as far as CMake is concerned -- but building
# Windows binaries from Linux is, and there the tools cannot run. Set
# SNP_CAPS_ARCHIVE to a prebuilt archive in that case.
include(CMakeDependentOption)
cmake_dependent_option(SNP_BUILD_CAPS_TOOLS
  "Build the host tools that generate caps.mpq (fetches StormLib)" ON
  "SNP_BUILD_PROVIDERS;NOT CMAKE_CROSSCOMPILING" OFF)

# Static CRT, everywhere, dependencies included. A module is dropped into a
# game directory on a machine we do not control, and every library linked into
# it has to agree on the runtime. The policy default reaches into fetched
# projects whose own cmake_minimum_required predates CMP0091.
#
# Only when this is the top-level project. A project that fetches the core --
# DevilutionX does -- has its own runtime, and a core built with another would
# not link into it.
if(MSVC AND PROJECT_IS_TOP_LEVEL)
  set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# --- Third-party code, before our own compile options -------------------
# Fetched projects become subdirectories of this one and inherit whatever
# directory options are set by then. Ours are strict and theirs are not ours.
if(SNPS_BUILD_TRANSPORT)
  include(LibDataChannel)
  snps_require_libdatachannel()
endif()
if(SNP_BUILD_CAPS_TOOLS)
  include(StormLib)
  snp_require_stormlib()
endif()

# --- Our own code --------------------------------------------------------
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra)
endif()

# The transport core builds for x86, x64 and wasm. Only the SNP adapter is
# 32-bit-only, and it enforces that itself.
add_subdirectory(core)

if(SNP_BUILD_CAPS_TOOLS)
  add_subdirectory(tools/mpq-builder)
endif()

add_subdirectory(adapters/snp)

# Needs snp_abi, and only makes sense where modules are built.
if(SNP_BUILD_PROVIDERS AND WIN32)
  add_subdirectory(tools/load-snp)
endif()

if(SNP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
